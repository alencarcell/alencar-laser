cmake_minimum_required(VERSION 3.21)
project(AlencarLaserPhase20d LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(alencar_phase20d_cache cache_probe.cpp)
target_link_libraries(alencar_phase20d_cache PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
if(MSVC)
 target_compile_options(alencar_phase20d_cache PRIVATE /O2 /W4 /utf-8)
else()
 target_compile_options(alencar_phase20d_cache PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
